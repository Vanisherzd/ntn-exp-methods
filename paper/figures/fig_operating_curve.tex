% Fig. 2 -- L4.7 operating characteristic. Included by paper/icc_main.tex via \input.
%
% AXIS NOTE. An earlier version plotted the data at x = 5*ICC (the effect SD) while ticking the axis
% in ICC, so the ticks labelled 0.5 and 0.8 sat at ICC 0.4 and 0.6 and the curve ran past the end of
% its own axis. The x coordinate is ICC itself. Keep it that way: if a point is added, give it an ICC
% coordinate, never an effect size.
%
% GEOMETRY NOTE. The previous version put a rotated "blind" label inside a narrow grey band hugging
% the y-axis, a three-line red annotation pressed against the same axis, and the alpha label sitting
% on the dashed line. Rendered at print size the band read as an artifact and the annotations
% competed with the curve. Now: the curve is the only heavy stroke, the low-power region is a
% bracket under the axis rather than a shaded block, the null callout sits away from both axis and
% curve, and the alpha label is at the right end just above its line.
%
% INDETERMINATE IS NOT AN ICC REGION. It depends on the design's grouping geometry, not on rho, so
% it is deliberately absent from this plot and stated in the caption instead. Never shade low ICC and
% label it INDETERMINATE.
%
% Every number is read from evaluation/results/l47_power_curve.json and the l47_calibration block of
% final_summary.json, and the three headline values are bound to those artifacts via \artv.
\begin{figure}[!t]
\centering
\resizebox{\columnwidth}{!}{%
\begin{tikzpicture}[font=\footnotesize,x=92mm,y=15mm]
% ---- axes

\draw[->,black!70] (0,0) -- (0,1.20);
\foreach \x/\lab in {0/{0}, 0.2/{0.2}, 0.4/{0.4}, 0.6/{0.6}, 0.8/{0.8}}

\foreach \y/\lab in {0/{0}, 0.5/{0.5}, 1/{1.0}}
  {\draw[black!70] (0,\y) -- (-0.007,\y); \node[left,font=\scriptsize] at (-0.009,\y) {\lab};}
\node[below,font=\scriptsize] at (0.50,-0.44)
  {intraclass correlation at the declared coarser level};
\node[rotate=90,font=\scriptsize] at (-0.075,0.6) {P(HALT)};

% ---- the power curve: the only heavy stroke in the figure.
% Every x is the icc field of a curve entry in l47_power_curve.json, every y its halt_prob, and every

% at 0.084 instead of 0.083, and drew the 0.01 entry into the line while omitting its marker.
\draw[line width=1.2pt,black!88]
  (0.002,0.05) -- (0.010,0.05) -- (0.038,0.05) -- (0.083,0.05)

\foreach \x/\y in {0.002/0.05, 0.010/0.05, 0.038/0.05, 0.083/0.05,

  {\fill[black!88] (\x,\y) circle (1.6pt);}

% ---- clean/null false-halt estimate, stated rather than plotted as a point.
% At y=15mm the whole Wilson interval spans 0.5mm, so the estimate, its interval, the alpha line and
% the flat end of the curve all fall inside one line width: rendered as a marker with error bars it
% was an unreadable red blob touching the leftmost curve marker. The numbers are the message and they
% stay bound to the artifact; only the false precision of a plotted position is dropped. Do not
% reinstate a plotted point without also giving the y axis enough height to resolve 0.031 from 0.05.
\draw[red!65!black,line width=0.4pt] (0.042,0.755) -- (0.062,0.115);
\node[font=\scriptsize,text=red!65!black,anchor=west,align=left,inner sep=1pt] at (0.045,0.80)

\caption{Operating characteristic of \rulename{L4.7} on the calibrated synthetic design (eight
coarser groups of three units). The clean/null false-halt estimate is $0.031$ with Wilson interval
$[0.018,0.052]$ against nominal $\alpha=0.05$---consistent with $\alpha$, not better; the two levels
differ by less than a line width at this scale, so the estimate is stated rather than plotted. Power rises
with dependence, reaching $1.00$ at $\rho=0.8$ but only $0.25$ at $\rho=0.2$, each over \num{40}
seeds, and approaching the false-halt rate for weak dependence. \textsc{indeterminate} is governed
separately by the attainable permutation resolution of the design---fewer than four coarser groups
or eight units---not by $\rho$.}
\label{fig:opcurve}
\end{figure}
