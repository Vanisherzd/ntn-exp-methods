% 8 ---------------------------------------------------------------- method as a procedure
\begin{frame}{The method is a procedure, not a taxonomy}
  \small
  \begin{enumerate}\itemsep0.55em
    \item \textbf{Classify} the validity obligation: is it row-local, or relational?
    \item \textbf{Choose} the comparison that makes a relational obligation falsifiable ---
          another execution, another source state, or one declared level coarser.
    \item \textbf{Encode} the necessary invariant and adjudicate
          \textbf{PASS} / \textbf{HALT} / \textbf{INDETERMINATE}.
    \item \textbf{Bind} the verdict and the manuscript claim to reproducible evidence.
  \end{enumerate}
  \msg{An advisor should be able to reproduce the method from this slide alone.}

% 10 --------------------------------------------------------------- L4.6 impossible problem
\begin{frame}{L4.6: an impossible-looking problem}
  % \relax first: beamer parses a brace group immediately after the frame title as the SUBTITLE
  % argument, so this opening sentence was silently swallowed and never reached the PDF.
  \relax
  {\small\textbf{How can a manifest prove it contains every behaviour-changing dependency?}}
  \vspace{0.7em}

  {\small It cannot directly: the omitted dependency is \emph{exactly} what is absent from the
   manifest.}
  \vspace{0.9em}

  {\small The brute-force response would be to enumerate every}
  \begin{itemize}\small\itemsep0.15em
    \item environment variable, cache, temporary file, on-disk state,
    \item external service, library build configuration, locale, clock source.
  \end{itemize}
  \vspace{0.4em}
  {\small\textcolor{oewarn}{Neither complete nor scalable} --- and an omission is invisible to it.}
  \msg{The problem is not lack of effort. It is that enumeration cannot certify its own coverage.}
\end{frame}

\begin{frame}{L4.6: the relational solution}
  % \relax first: beamer parses a brace group immediately after the frame title as the SUBTITLE
  % argument, so this opening sentence was silently swallowed and never reached the PDF.
  \relax
  % Intuition BEFORE notation. This slide used to open on the implication, so an advisor met the
  % quantified form before ever seeing the two runs it describes.
  {\small Run the pipeline twice under controlled variation:}

  \begin{center}\scriptsize
  \begin{tabular}{@{}lll@{}}
    \toprule
    & declared manifest & output\\
    \midrule
    run A & hash $H$ & $X$\\
    run B & hash $H$ & $Y$\\
    \bottomrule
  \end{tabular}
  \end{center}

  {\small $X \neq Y$ while the manifest is \emph{identical} $\Rightarrow$ \textbf{the declared
   provenance is incomplete}: something outside it changed the behaviour.}
  \vspace{0.3em}

  {\footnotesize \textbf{Failure} is a constructive counterexample. \textbf{Success} is
   \textcolor{oewarn}{not a certificate} --- agreement is consistent with completeness \emph{and}
   with an untested dependency. Precondition and quantified form: A7, A2.}
  % \par first, for the same reason as slide 4: without it the one-message takeaway was set inline
  % at the end of the dense paragraph above and stopped reading as the slide's conclusion.
  \par\vspace{0.45em}
  {\small We do not prove completeness --- we make \textbf{incompleteness falsifiable}.}
  \notclaim{that L4.6 verifies completeness. It refutes a \emph{claim} of completeness.}
\end{frame}

% 13 --------------------------------------------------------------- L4.7 the gate
\begin{frame}{L4.7: the statistical-unit gate}
  % \relax first: beamer parses a brace group immediately after the frame title as the SUBTITLE
  % argument, so this opening sentence was silently swallowed and never reached the PDF.
  \relax
  {\small Values $y$, declared units $u$, one coarser physical grouping $g$. Aggregate to unit
   values, then measure residual dependence \emph{within} $g$.}
  \vspace{0.4em}
  \[ p = \frac{1 + \#\{\hat{\rho}^{*} \ge \hat{\rho}\}}{B+1},
     \qquad B = \NpermB. \]
  \vspace{-0.5em}
  \begin{itemize}\small\itemsep0.2em
    \item $\hat{\rho}$ is a one-way ICC(1) \srcref{Shrout \& Fleiss 1979} used as an \emph{effect statistic}.
    \item The \emph{decision} is referenced to a permutation null \srcref{Winkler et al.\ 2014} over the declared grouping.
  \end{itemize}
  \vspace{0.3em}
  {\small ICC(1) is not new. Permutation inference is not new. The novelty is using this
   calibrated decision as an \textbf{executable experiment-validity gate}.}
\end{frame}

% 14 --------------------------------------------------------------- why the gate can abstain
\begin{frame}{Why the gate is allowed to abstain}
  \small
  \begin{tabular}{@{}lrr@{}}
    \toprule
    design & distinct assignments & smallest attainable $p$\\
    \midrule
    3 coarser groups $\times$ 2 units & \Nassignthree & \Nminpthree\\
    4 coarser groups $\times$ 2 units & \Nassignfour  & \Nminpfour\\
    \bottomrule
  \end{tabular}
  \vspace{0.7em}

  At \Nassignthree{} assignments the smallest attainable $p$ is \Nminpthree{}, which exceeds
  $\alpha=\Nalpha$: that design \textbf{cannot reject at any effect size}. The rule returns
  \textbf{INDETERMINATE} rather than PASS.
  \vspace{0.4em}

  {\small\textbf{Attainability floor $\neq$ low power.} Low power is about $\rho$; this is about
   what the grouping geometry can resolve at all.}
  \notclaim{that insufficient decision resolution is evidence of validity.}
\end{frame}

\begin{frame}{Every represented fault fires its rule; chronology catches two}

  \begin{tabular}{@{}p{0.60\textwidth}r@{}}

  \notclaim{unseen-fault sensitivity. \Ncontract/\Nfaults{} is \textbf{represented-fault
   regression reachability}, not detector accuracy.}

\begin{frame}{A2 --- The L4.7 permutation reference}
  \small
  Under the null the coarser grouping carries \emph{no} information, so unit values are
  exchangeable \textbf{across} the groups of $g$. The reference $\{\hat{\rho}^{*}\}$ comes from
  permuting unit values freely over \emph{all} units while the group memberships and sizes stay
  fixed --- so each draw keeps the observed group count and balance.
  \vspace{0.3em}

  {\footnotesize Permuting \emph{within} a group would be a no-op --- composition unchanged,
   ICC(1) invariant, every draw equal to $\hat{\rho}$.}
  \[ p = \frac{1 + \#\{\hat{\rho}^{*} \ge \hat{\rho}\}}{B+1}, \qquad B = \NpermB. \]
  {\footnotesize The $+1$ on both sides is the finite-$B$ correction: it counts the observed
   statistic as one of its own reference draws, so the test is valid at any $B$ and the smallest
   reportable $p$ is $1/(\NpermB+1)$ rather than $0$. \emph{Every} real-data HALT sits exactly there.
   \vspace{0.25em}

   The numerator is unbiased, so the estimate can go negative and is reported truncated at zero.
   It uses the \emph{mean} group size, so on unbalanced data (\Natused{} units in \Natelsets{}
   groups) the estimate is a balanced-design approximation --- permutation referencing keeps the
   \emph{decision} size-valid either way.}
\end{frame}

\begin{frame}{A3 --- Attainability versus power}
  \small
  \begin{tabular}{@{}lrrl@{}}
    \toprule
    design & assignments & smallest \emph{attainable} $p$ & can reach $\alpha=\Nalpha$?\\
    \midrule
    3 groups $\times$ 2 units & \Nassignthree & \Nminpthree & \textcolor{oewarn}{no --- at any effect size}\\
    4 groups $\times$ 2 units & \Nassignfour  & \Nminpfour  & yes\\
    \bottomrule
  \end{tabular}
  \vspace{0.6em}

  Assignments for $k$ unlabelled groups of $m$ units: $(km)!/(m!^{k}k!)$ --- arithmetic about the
  \emph{design}, evaluated before any data is seen, which is why the precondition is asserted
  rather than estimated.
  \vspace{0.45em}

  \textbf{Two floors, two questions.} The column above is combinatorial: the smallest $p$ that
  \emph{any} permutation reference could attain for that design. It answers ``can this design
  reach $\alpha$ at all?'' --- and at \Nassignthree{} assignments it cannot. The implemented rule
  is separate: it always draws $B=\NpermB$ random permutations, so the finest $p$ it
  \emph{reports} is $1/(\NpermB+1)$. The precondition excludes only designs whose combinatorial
  floor already exceeds $\alpha$ --- it declines to run below \Nmingroups{} coarser groups or
  \Nminunits{} units, which is exactly the \Nassignthree-assignment row.
\end{frame}
